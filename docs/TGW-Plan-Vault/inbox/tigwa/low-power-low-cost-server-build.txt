Understanding the strengths and limitations of both formats is crucial for optimizing your AI development environment.

    Graphics Power: The absolute biggest advantage. You can fit high-end desktop GPUs like the Nvidia RTX 4080 or 4090, delivering unmatched processing speeds for PyTorch workflows and CUDA-accelerated tasks.

The Unified NixOS Configuration Snippet
Add this logic to your /etc/nixos/configuration.nix file to configure your environment:
nix

{ config, pkgs, ... }:

{
  # 1. ALLOW UNFREE PACKAGES (Mandatory for Nvidia CUDA)
  nixpkgs.config.allowUnfree = true;

  # 2. ENABLE AMD INTEGRATED GRAPHICS & XSERVER/WAYLAND DRIVERS
  # This makes the 7600 iGPU run your desktop, freeing up 100% of the RTX VRAM.
  services.xserver.videoDrivers = [ "nvidia" ];

  hardware.graphics = {
    enable = true;
    enable32Bit = true;
  };

  # 3. NVIDIA CONGIGURATION
  hardware.nvidia = {
    modesetting.enable = true;
    # Essential for headless microVM/Docker/Inference tasks to keep drivers active
    powerManagement.enable = true;
    open = false; # Set to true if you prefer the open-source kernel module
    package = config.boot.kernelPackages.nvidiaPackages.stable;
  };

  # 4. SYSTEMD SERVICE: THE LOW-POWER HEAT-KILLER (Sets GPU to 112W)
  # nvidia-smi resets power limits on reboot. This systemd hook fixes that.
  systemd.services.nvidia-power-limit = {
    description = "Set Nvidia RTX 4060 Ti Power Limit to 112W";
    after = [ "nvidia-persistenced.service" ];
    wants = [ "nvidia-persistenced.service" ];
    wantedBy = [ "multi-user.target" ];
    serviceConfig = {
      Type = "oneshot";
      RemainAfterExit = true;
      # Enables Persistence Mode first, then enforces the 112W limit (70% of 160W)
      ExecStart = ''
        ${config.boot.kernelPackages.nvidiaPackages.stable}/bin/nvidia-smi -pm 1
        ${config.boot.kernelPackages.nvidiaPackages.stable}/bin/nvidia-smi -pl 112
      '';
    };
  };

  # 5. DOCKER & CONTAINER TOOLKIT WITH NATIVE GPU PASS-THROUGH
  # Ensures your dockerized local models (Ollama, PyTorch containers) see the GPU.
  virtualisation.docker = {
    enable = true;
    enableOnBoot = true;
  };

  # Modern declarative approach to expose Nvidia layers to Docker/Podman
  hardware.nvidia-container-toolkit.enable = true;
}
